# Local development configuration for the Admin & Teacher Dashboard.
#
# Generated for this machine. Not committed — .gitignore excludes .env*.local.

# The NestJS API, including its global prefix. The backend's .env sets
# PORT=3000, API_PREFIX=api and API_VERSION=1, which it serves as /api/v1.
API_BASE_URL=http://localhost:3000/api/v1

# How long to wait on the API for a single request.
API_TIMEOUT_MS=20000

# MUST stay false over plain HTTP. A `Secure` cookie is silently dropped by the
# browser on http://, and the symptom is a login that appears to do nothing.
# Set this to true (or remove it) the moment the dashboard is served over HTTPS.
SECURE_COOKIES=false

# The dashboard's own origin. The proxy refuses mutating requests that arrive
# claiming any other origin. `npm run dev` and `npm start` both use port 3001,
# which is already in the backend's CORS_ORIGINS list.
APP_ORIGIN=http://localhost:3001

# Every date in the interface is rendered in this zone. Keep the two in step.
PLATFORM_TIMEZONE=Africa/Cairo
NEXT_PUBLIC_PLATFORM_TIMEZONE=Africa/Cairo
